"""
文档内提取双星结构，在原有的基础上扩展多跳节点，最大跳数为5
如果返回的重复实体出现次数正好为2，则其出现的两个三元组分别作为双星的两个星型，这个重复实体作为两个星型各自的中心，并且该重复实体用虚拟连接边连接
如果返回的重复实体出现次数大于等于3，则借助三元组中"edge_properties"作为聚类函数进行聚类，将所有该重复实体出现的三元组按照距离远近聚成两类，分别作为两个双星，然后该重复实体作为两个双星的中心，并按照虚拟连接边连接
"""

def generate_virtual_links(star_uids):
    """
    生成虚拟边，连接所有星型结构的中心节点，表示同一实体
    """
    virtual_edges = []
    for i in range(len(star_uids)):
        for j in range(i + 1, len(star_uids)):
            virtual_edges.append({
                "node_1": {"name": star_uids[i], "label": "VirtualCenter"},
                "node_2": {"name": star_uids[j], "label": "VirtualCenter"},
                "relationship": "same_entity"
            })
    return virtual_edges

def extract_bipartite_stars(graph, central_entities):
    """
    提取双星结构：
    - 如果实体出现次数为2，直接作为双星结构
    - 如果实体出现次数>=3，使用KMeans进行聚类
    """
    stars = []

    for entity, count in central_entities.items():
        # 获取该实体的所有三元组
        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        if count == 2:
            # 如果实体出现次数为2，直接构成双星结构
            star1 = {
                'center': entity,
                'triples': triples[:1],  # 第一组三元组
            }
            star2 = {
                'center': entity,
                'triples': triples[1:],  # 第二组三元组
            }
            # 创建虚拟连接边
            virtual_links = generate_virtual_links([entity])
            stars.append((star1, star2, virtual_links))
        elif count >= 3:
            # 如果实体出现次数>=3，使用KMeans进行聚类
            edge_properties = [entry['edge_properties'] for entry in triples]

            # 提取数值特征：假设edge_properties包含 'text_span' 字段
            X = []
            for prop in edge_properties:
                # 假设prop['text_span']包含 'start' 和 'end'
                text_span = prop.get('text_span', {})
                start = text_span.get('start', 0)  # 默认值为0
                end = text_span.get('end', 0)  # 默认值为0

                # 使用 (start + end) / 2 计算质心
                centroid = (start + end) / 2

                # 使用质心作为特征
                X.append([centroid])

            X = np.array(X)

            # 使用KMeans聚类为2类
            kmeans = KMeans(n_clusters=2, random_state=0).fit(X)
            labels = kmeans.labels_

            star1 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 0],  # 第一类三元组
            }
            star2 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 1],  # 第二类三元组
            }
            # 创建虚拟连接边
            virtual_links = generate_virtual_links([entity])
            stars.append((star1, star2, virtual_links))

    return stars

def simplify_edge_labels_with_llm(relationships, llm):
    system_prompt = (
        "You are a helpful assistant. Your task is to simplify relationship phrases "
        "into short, clear, lowercase labels (e.g., 'works_with', 'married_to', 'located_in'). "
        "Avoid full sentences or long phrases. Use underscores to separate words."
    )

    joined = "\n".join(f"- {rel}" for rel in relationships)
    user_prompt = f"""Simplify the following relationship phrases:

{joined}

Return one simplified label per line in the same order."""

    response = llm.generate(user_message=user_prompt, system_message=system_prompt)
    simplified = [line.strip("- ").strip() for line in response.strip().splitlines() if line.strip()]
    return simplified


def visualize_property_graph_with_simplified_edges(json_data, llm, title="Knowledge Graph", output_file="graph.png"):
    """
    Visualizes the property graph with simplified edge labels.
    """
    relationships = set()

    # Iterate over each star structure in the json_data
    for star in json_data:
        for entry in star['triples']:
            if "relationship" in entry:
                relationships.add(entry["relationship"])
            else:
                print(f"Warning: Missing 'relationship' key in entry: {entry}")

    # Convert to list for further processing
    relationships = list(relationships)

    if not relationships:
        print("Error: No valid relationships found in the input data.")
        return  # Exit the function if no relationships are found

    simplified_labels = simplify_edge_labels_with_llm(relationships, llm)
    rel_map = dict(zip(relationships, simplified_labels))

    G = nx.DiGraph()
    label_colors = {
        "Person": "#87CEFA", "Company": "#FFD700", "Event": "#FFB6C1",
        "Department": "#98FB98", "Location": "#DDA0DD", "default": "#D3D3D3"
    }

    # Iterate over each star and its triples to construct the graph
    for star in json_data:
        for entry in star['triples']:
            if "relationship" in entry:
                node1 = copy.deepcopy(entry["node_1"])
                node2 = copy.deepcopy(entry["node_2"])

                name1 = node1.pop("name", node1.pop("label", "Unknown"))
                name2 = node2.pop("name", node2.pop("label", "Unknown"))
                label1 = node1.get("label", "default")
                label2 = node2.get("label", "default")

                G.add_node(name1, label=label1)
                G.add_node(name2, label=label2)

                simplified_label = rel_map.get(entry["relationship"], "related")
                G.add_edge(name1, name2, label=simplified_label, orig_rel=entry["relationship"])

    pos = None
    if "Dual-Star Structure:" in title:
        centers = [n for n, d in G.nodes(data=True) if d.get("label") == "VirtualCenter"]
        if len(centers) >= 2:
            center1, center2 = centers[0], centers[1]
            group1 = set([center1])
            group2 = set([center2])

            for nbr in G.neighbors(center1):
                group1.add(nbr)
            for nbr in G.neighbors(center2):
                group2.add(nbr)

            pos = {}
            for i, node in enumerate(group1):
                pos[node] = (0, -i)
            for i, node in enumerate(group2):
                pos[node] = (3, -i)

            others = set(G.nodes()) - group1 - group2
            if others:
                mid_pos = nx.spring_layout(G.subgraph(others), seed=42)
                for node, p in mid_pos.items():
                    pos[node] = (1.5, p[1])
        else:
            pos = nx.spring_layout(G, k=1.2, seed=42)
    else:
        pos = nx.spring_layout(G, k=1.2, seed=42)

    fig, ax = plt.subplots(figsize=(20, 14))
    node_colors = [label_colors.get(G.nodes[n].get("label", "default"), "#D3D3D3") for n in G.nodes()]
    nx.draw_networkx_nodes(G, pos, node_color=node_colors, node_size=1200, edgecolors="black", ax=ax)

    node_labels = {node: node for node in G.nodes()}
    nx.draw_networkx_labels(G, pos, labels=node_labels, font_size=9, ax=ax)

    non_same_entity_edges = [(u, v) for u, v, d in G.edges(data=True) if d.get("orig_rel") != "same_entity"]
    nx.draw_networkx_edges(G, pos, edgelist=non_same_entity_edges, arrowstyle='-|>', arrowsize=40, edge_color='gray', ax=ax)

    same_entity_edges = [(u, v) for u, v, d in G.edges(data=True) if d.get("orig_rel") == "same_entity"]
    nx.draw_networkx_edges(G, pos, edgelist=same_entity_edges, arrowstyle='<->', arrowsize=45, edge_color='red', width=2, ax=ax)

    edge_labels = {(u, v): d.get("label", "") for u, v, d in G.edges(data=True)}
    nx.draw_networkx_edge_labels(G, pos, edge_labels=edge_labels, font_size=7, font_color='darkred', ax=ax)

    plt.title(title, fontsize=18)
    plt.axis("off")
    plt.savefig(output_file, dpi=300)
    plt.close()
    print(f"[SAVED] 图已保存为: {output_file}")


def count_node_1_entities(graph):
    # 提取所有node_1的实体名称
    node_1_entities = [entry['node_1']['name'] for entry in graph]

    # 使用Counter统计每个实体出现的次数
    entity_counts = Counter(node_1_entities)

    # 过滤出出现次数大于等于2的实体
    central_entities = {entity: count for entity, count in entity_counts.items() if count >= 2}

    return central_entities

    with open(file_path, 'r', encoding='utf-8') as file:
        data = json.load(file)
    return data


def remove_single_hop_neighbors(star):
    """ 删除多跳邻居中的跳数为1的邻居，并删除与 neighbors 中节点重复的 multi_hop_neighbors 节点 """

    # 删除 hop_count 为 1 的邻居
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"] if neighbor_info["hop_count"] > 1
    ]

    # 获取 neighbors 中的所有节点
    neighbor_nodes = set(star["neighbors"])

    # 遍历 multi_hop_neighbors 删除重复的节点
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"]
        if neighbor_info["neighbor"] not in neighbor_nodes
    ]

    # 返回修改后的星型结构
    return star



def get_multi_hop_neighbors(node, graph, max_hops, current_hop=1, visited=set(), multi_hop_neighbors=[]):
    """
    递归获取某个节点的多跳邻居，最多为 max_hops 跳。
    """
    if current_hop > max_hops:
        return visited, multi_hop_neighbors

    visited.add(node)

    # 查找该节点的所有三元组，找到与节点连接的其他节点
    for entry in graph:
        if entry['node_1']['name'] == node:
            neighbor = entry['node_2']['name']
            relationship = entry['relationship']
        elif entry['node_2']['name'] == node:
            neighbor = entry['node_1']['name']
            relationship = entry['relationship']
        else:
            continue

        # 如果邻居没有访问过，则递归查找该邻居的多跳邻居
        if neighbor not in visited:
            # 如果没有 'edge' 字段，设置一个默认值
            edge = entry.get('relationship', 'unknown_edge')  # 使用关系作为 edge
            multi_hop_neighbors.append({
                "neighbor": neighbor,
                "hop_count": current_hop,
                "relationship": relationship
            })
            visited, multi_hop_neighbors = get_multi_hop_neighbors(neighbor, graph, max_hops, current_hop + 1, visited, multi_hop_neighbors)

    return visited, multi_hop_neighbors



def modify_center_and_node(center, triples, suffix):
    """
    修改 center 和 node_1.name 中的字段值，给所有字段的值添加后缀（_1 或 _2）
    """
    # 如果center是字符串，将其值后加后缀
    if isinstance(center, str):
        center = center + "_" + suffix

    # 修改每个三元组中的node_1.name
    for triple in triples:
        if isinstance(triple['node_1'], dict) and 'name' in triple['node_1']:
            triple['node_1']['name'] = triple['node_1']['name'] + "_" + suffix

    return center, triples

def extend_bipartite_stars_with_hops(graph, central_entities, max_hops=5):
    """
    提取双星结构并延伸多跳邻居
    """
    stars = []

    for entity, count in central_entities.items():
        # 获取该实体的所有三元组
        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        # 1跳邻居（基础的双星结构）
        star1 = {
            'center': entity,
            'triples': triples[:1],  # 第一组三元组
            'neighbors': [entry['node_2']['name'] for entry in triples],
            'edges': [entry['relationship'] for entry in triples],
            'multi_hop_neighbors': []  # 初始化为空列表
        }

        star2 = {
            'center': entity,
            'triples': triples[1:],  # 第二组三元组
            'neighbors': [entry['node_2']['name'] for entry in triples],
            'edges': [entry['relationship'] for entry in triples],
            'multi_hop_neighbors': []  # 初始化为空列表
        }

        # 修改 center 和 node_1.name 中的字段值，添加后缀 _1 和 _2
        star1['center'], star1['triples'] = modify_center_and_node(star1['center'], star1['triples'], '1')
        star2['center'], star2['triples'] = modify_center_and_node(star2['center'], star2['triples'], '2')

        # 计算多跳邻居
        visited, multi_hop_neighbors = get_multi_hop_neighbors(entity, graph, max_hops)

        # 确保多跳邻居是列表，其中的每个元素是字典
        star1['multi_hop_neighbors'] = multi_hop_neighbors  # 直接赋值
        star2['multi_hop_neighbors'] = multi_hop_neighbors  # 直接赋值

        # 删除单跳邻居
        star1 = remove_single_hop_neighbors(star1)
        star2 = remove_single_hop_neighbors(star2)

        # 创建虚拟连接边
        virtual_links = generate_virtual_links([entity])

        # 将双星结构添加到结果列表中
        stars.append((star1, star2, virtual_links))

    return stars




def visualize_star_structure_with_hops(json_data, output_file, title="Star Structure", llm=None):
    """ 可视化包含多跳邻居的星型结构，并限制下一跳节点只能作为出边，同时将关系放置在侧边栏 """
    # 检查 json_data 是否是字典类型，并提取中心节点、邻居节点、边和多跳邻居
    if isinstance(json_data, dict):
        center = json_data.get("center")
        neighbors = json_data.get("neighbors", [])
        edges = json_data.get("edges", [])
        multi_hop_neighbors = json_data.get("multi_hop_neighbors", [])
    else:
        print("Error: json_data should be a dictionary with 'center', 'neighbors', and 'multi_hop_neighbors' keys.")
        return

    if not center or not neighbors or not edges:
        print("Error: Missing required data in json_data. Ensure 'center', 'neighbors', and 'edges' are present.")
        return

    # 使用 LLM 简化边的标签（如果提供了 LLM）
    if llm:
        simplified_edge_labels = simplify_edge_labels_with_llm(edges, llm)
    else:
        simplified_edge_labels = edges  # 如果没有 LLM，则使用原始边标签

    # 创建一个有向图
    G = nx.DiGraph()

    # 添加中心节点和邻居节点
    G.add_node(center, label="center")
    for neighbor in neighbors:
        G.add_node(neighbor, label="neighbor")

    # 添加边，假设每条边的关系为边标签
    for simplified_edge, neighbor in zip(simplified_edge_labels, neighbors):
        G.add_edge(center, neighbor, label=simplified_edge)

    # 记录多跳邻居的前一跳邻居
    prev_neighbors = neighbors  # 初始化为直接邻居节点
    all_hop_neighbors = {0: neighbors}  # 用于存储各层级的邻居

    # 添加多跳邻居并与前一跳邻居相连，确保下一跳节点仅为出边
    for multi_hop in multi_hop_neighbors:
        # 使用字典的键来访问 neighbor、hop_count 和 relationship
        neighbor = multi_hop["neighbor"]
        hop_count = multi_hop["hop_count"]
        relationship = multi_hop["relationship"]

        # 推导出上一跳邻居（基于当前 hop_count 和前一跳的邻居）
        if hop_count > 1 and hop_count - 1 in all_hop_neighbors:
            previous_hop_neighbor = all_hop_neighbors[hop_count - 1]
        else:
            previous_hop_neighbor = neighbors  # 如果是第一跳，则直接连到中心节点

        # 新增多跳邻居节点
        G.add_node(neighbor, label=f"hop_{hop_count}")

        # 确保多跳邻居与前一跳邻居相连，但不允许反向连接
        if hop_count - 1 in all_hop_neighbors:
            prev_neighbors = all_hop_neighbors[hop_count - 1]
            for prev_neighbor in prev_neighbors:
                # 只添加出边，不添加入边
                if not G.has_edge(prev_neighbor, neighbor):  # 确保没有重复的边
                    G.add_edge(prev_neighbor, neighbor, label=relationship)

        # 连接上一跳邻居到当前多跳邻居
        if previous_hop_neighbor != center:  # 确保上一跳不是中心节点
            for prev_hop in previous_hop_neighbor:
                if not G.has_edge(prev_hop, neighbor):
                    G.add_edge(prev_hop, neighbor, label=relationship)

        # 更新各层级邻居
        if hop_count not in all_hop_neighbors:
            all_hop_neighbors[hop_count] = []
        all_hop_neighbors[hop_count].append(neighbor)

    # 设置图形布局
    pos = nx.spring_layout(G, k=1.2)
    fig, ax = plt.subplots(figsize=(20, 14))

    # 节点颜色
    label_colors = {
        "center": "#FFD700",  # 黄金色
        "neighbor": "#87CEFA",  # 淡蓝色
        "multi_hop_neighbor": "#32CD32",  # 多跳邻居，绿色
        "default": "#D3D3D3"  # 默认灰色
    }

    # 节点颜色映射
    node_colors = [
        label_colors.get(G.nodes[n].get("label", "default"), "#D3D3D3") for n in G.nodes()
    ]
    nx.draw_networkx_nodes(G, pos, node_color=node_colors, node_size=1200, edgecolors="black", ax=ax)

    # 绘制节点标签
    node_labels = {node: node for node in G.nodes()}
    nx.draw_networkx_labels(G, pos, labels=node_labels, font_size=9, ax=ax)

    # 绘制边
    nx.draw_networkx_edges(G, pos, arrowstyle='-|>', arrowsize=50, edge_color='gray', ax=ax)

    # 绘制边标签（我们将这些关系移到侧边栏）
    edge_labels = {(u, v): d.get("label", "") for u, v, d in G.edges(data=True)}

    # 计算侧边栏位置
    sidebar_x = 1.05  # 侧边栏x位置
    sidebar_y_start = 0.95  # 侧边栏y起始位置
    label_margin = 0.02  # 标签的垂直间距
    max_label_height = 0.85  # 侧边栏最大高度

    # 添加关系标签到侧边栏
    for (u, v), label in edge_labels.items():
        if label:  # 如果有标签，绘制在侧边栏
            ax.text(
                sidebar_x, sidebar_y_start, label, fontsize=8, ha='left', va='top',
                transform=ax.transAxes, color="darkred"
            )
            sidebar_y_start -= label_margin  # 每个标签下移

    # 设置标题
    plt.title(title, fontsize=18)
    plt.axis("off")

    # 确保创建输出目录
    output_dir = os.path.dirname(output_file)
    os.makedirs(output_dir, exist_ok=True)  # 确保目标目录存在

    # 保存图像
    plt.savefig(output_file, dpi=300, bbox_inches='tight')
    plt.close()
    print(f"[SAVED] 图已保存为: {output_file}")




def save_and_visualize_dual_star_with_hops(dual_star, virtual_links, output_dir):
    """
    保存双星结构为 JSON 文件，并可视化双星结构（包括多跳邻居）
    """
    safe_uid1 = dual_star[0]['center'].replace(" ", "_").replace("/", "_")
    safe_uid2 = dual_star[1]['center'].replace(" ", "_").replace("/", "_")

    json_filename = f"{output_dir}/dual_star_{safe_uid1}_vs_{safe_uid2}_multi_hop.json"

    # 合并 star1、star2 和虚拟边
    combined = dual_star[0]['triples'] + dual_star[1]['triples'] + [{
        "node_1": {"name": safe_uid1, "label": "VirtualCenter"},

    }]

    full_dual_star = {

    print(f"保存了双星结构为 JSON: {json_filename}")

    # 使用新的可视化方法展示双星结构（包括多跳邻居）
    for star in full_dual_star['stars']:
        center_name = star['center']
        output_image = f"{output_dir}/dual_star_{safe_uid1}_vs_{safe_uid2}_multi_hop.png"

        visualize_star_structure_with_hops(star, output_image, title, llm)
        print(f"双星结构图像已保存为 PNG: {output_image}")


def main(input_file, output_dir, max_hops):
    # 确保输出目录存在
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    # 从文件加载数据
    merged_enriched_graph = load_json_data(input_file)

    # 获取重复实体
    central_entities = count_node_1_entities(merged_enriched_graph)

    # 提取双星结构并延伸多跳邻居
    bipartite_stars_with_virtual_links = extend_bipartite_stars_with_hops(merged_enriched_graph, central_entities, max_hops)

    # 打印并保存双星结构
    for idx, (star1, star2, virtual_links) in enumerate(bipartite_stars_with_virtual_links):
        print(f"双星结构 {idx + 1}:")

def generate_virtual_links(star_uids):
    """
    生成虚拟边，连接所有星型结构的中心节点，表示同一实体
    """
    virtual_edges = []
    for i in range(len(star_uids)):
        for j in range(i + 1, len(star_uids)):
            virtual_edges.append({
                "node_1": {"name": star_uids[i], "label": "VirtualCenter"},
                "node_2": {"name": star_uids[j], "label": "VirtualCenter"},
                "relationship": "same_entity"
            })
    return virtual_edges

def extract_bipartite_stars(graph, central_entities):
    """
    提取双星结构：
    - 如果实体出现次数为2，直接作为双星结构
    - 如果实体出现次数>=3，使用KMeans进行聚类
    """
    stars = []

    for entity, count in central_entities.items():
        # 获取该实体的所有三元组
        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        if count == 2:
            # 如果实体出现次数为2，直接构成双星结构
            star1 = {
                'center': entity,
                'triples': triples[:1],  # 第一组三元组
            }
            star2 = {
                'center': entity,
                'triples': triples[1:],  # 第二组三元组
            }
            # 创建虚拟连接边
            virtual_links = generate_virtual_links([entity])
            stars.append((star1, star2, virtual_links))
        elif count >= 3:
            # 如果实体出现次数>=3，使用KMeans进行聚类
            edge_properties = [entry['edge_properties'] for entry in triples]

            # 提取数值特征：假设edge_properties包含 'text_span' 字段
            X = []
            for prop in edge_properties:
                # 假设prop['text_span']包含 'start' 和 'end'
                text_span = prop.get('text_span', {})
                start = text_span.get('start', 0)  # 默认值为0
                end = text_span.get('end', 0)  # 默认值为0

                # 使用 (start + end) / 2 计算质心
                centroid = (start + end) / 2

                # 使用质心作为特征
                X.append([centroid])

            X = np.array(X)

            # 使用KMeans聚类为2类
            kmeans = KMeans(n_clusters=2, random_state=0).fit(X)
            labels = kmeans.labels_

            star1 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 0],  # 第一类三元组
            }
            star2 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 1],  # 第二类三元组
            }
            # 创建虚拟连接边
            virtual_links = generate_virtual_links([entity])
            stars.append((star1, star2, virtual_links))

    return stars

def simplify_edge_labels_with_llm(relationships, llm):
    system_prompt = (
        "You are a helpful assistant. Your task is to simplify relationship phrases "
        "into short, clear, lowercase labels (e.g., 'works_with', 'married_to', 'located_in'). "
        "Avoid full sentences or long phrases. Use underscores to separate words."
    )

    joined = "\n".join(f"- {rel}" for rel in relationships)
    user_prompt = f"""Simplify the following relationship phrases:

{joined}

Return one simplified label per line in the same order."""

    response = llm.generate(user_message=user_prompt, system_message=system_prompt)
    simplified = [line.strip("- ").strip() for line in response.strip().splitlines() if line.strip()]
    return simplified

def remove_single_hop_neighbors(star):
    """ 删除多跳邻居中的跳数为1的邻居，并删除与 neighbors 中节点重复的 multi_hop_neighbors 节点 """

    # 删除 hop_count 为 1 的邻居
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"] if neighbor_info["hop_count"] > 1
    ]

    # 获取 neighbors 中的所有节点
    neighbor_nodes = set(star["neighbors"])

    # 遍历 multi_hop_neighbors 删除重复的节点
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"]
        if neighbor_info["neighbor"] not in neighbor_nodes
    ]

    # 返回修改后的星型结构
    return star



def get_multi_hop_neighbors(node, graph, max_hops, current_hop=1, visited=set(), multi_hop_neighbors=[]):
    """
    递归获取某个节点的多跳邻居，最多为 max_hops 跳。
    """
    if current_hop > max_hops:
        return visited, multi_hop_neighbors

    visited.add(node)

    # 查找该节点的所有三元组，找到与节点连接的其他节点
    for entry in graph:
        if entry['node_1']['name'] == node:
            neighbor = entry['node_2']['name']
            relationship = entry['relationship']
        elif entry['node_2']['name'] == node:
            neighbor = entry['node_1']['name']
            relationship = entry['relationship']
        else:
            continue

        # 如果邻居没有访问过，则递归查找该邻居的多跳邻居
        if neighbor not in visited:
            # 如果没有 'edge' 字段，设置一个默认值
            edge = entry.get('relationship', 'unknown_edge')  # 使用关系作为 edge
            multi_hop_neighbors.append({
                "neighbor": neighbor,
                "hop_count": current_hop,
                "relationship": relationship
            })
            visited, multi_hop_neighbors = get_multi_hop_neighbors(neighbor, graph, max_hops, current_hop + 1, visited, multi_hop_neighbors)

    return visited, multi_hop_neighbors

def visualize_star_structure_with_hops(json_data, output_file, title="Star Structure", llm=None):
    """ 可视化包含多跳邻居的星型结构，并限制下一跳节点只能作为出边，同时将关系放置在侧边栏 """
    # 检查 json_data 是否是字典类型，并提取中心节点、邻居节点、边和多跳邻居
    if isinstance(json_data, dict):
        center = json_data.get("center")
        neighbors = json_data.get("neighbors", [])
        edges = json_data.get("edges", [])
        multi_hop_neighbors = json_data.get("multi_hop_neighbors", [])
    else:
        print("Error: json_data should be a dictionary with 'center', 'neighbors', and 'multi_hop_neighbors' keys.")
        return

    if not center or not neighbors or not edges:
        print("Error: Missing required data in json_data. Ensure 'center', 'neighbors', and 'edges' are present.")
        return

    # 使用 LLM 简化边的标签（如果提供了 LLM）
    if llm:
        simplified_edge_labels = simplify_edge_labels_with_llm(edges, llm)
    else:
        simplified_edge_labels = edges  # 如果没有 LLM，则使用原始边标签

    # 创建一个有向图
    G = nx.DiGraph()

    # 添加中心节点和邻居节点
    G.add_node(center, label="center")
    for neighbor in neighbors:
        G.add_node(neighbor, label="neighbor")

    # 添加边，假设每条边的关系为边标签
    for simplified_edge, neighbor in zip(simplified_edge_labels, neighbors):
        G.add_edge(center, neighbor, label=simplified_edge)

    # 记录多跳邻居的前一跳邻居
    prev_neighbors = neighbors  # 初始化为直接邻居节点
    all_hop_neighbors = {0: neighbors}  # 用于存储各层级的邻居

    # 添加多跳邻居并与前一跳邻居相连，确保下一跳节点仅为出边
    for multi_hop in multi_hop_neighbors:
        # 使用字典的键来访问 neighbor、hop_count 和 relationship
        neighbor = multi_hop["neighbor"]
        hop_count = multi_hop["hop_count"]
        relationship = multi_hop["relationship"]

        # 推导出上一跳邻居（基于当前 hop_count 和前一跳的邻居）
        if hop_count > 1 and hop_count - 1 in all_hop_neighbors:
            previous_hop_neighbor = all_hop_neighbors[hop_count - 1]
        else:
            previous_hop_neighbor = neighbors  # 如果是第一跳，则直接连到中心节点

        # 新增多跳邻居节点
        G.add_node(neighbor, label=f"hop_{hop_count}")

        # 确保多跳邻居与前一跳邻居相连，但不允许反向连接
        if hop_count - 1 in all_hop_neighbors:
            prev_neighbors = all_hop_neighbors[hop_count - 1]
            for prev_neighbor in prev_neighbors:
                # 只添加出边，不添加入边
                if not G.has_edge(prev_neighbor, neighbor):  # 确保没有重复的边
                    G.add_edge(prev_neighbor, neighbor, label=relationship)

        # 连接上一跳邻居到当前多跳邻居
        if previous_hop_neighbor != center:  # 确保上一跳不是中心节点
            for prev_hop in previous_hop_neighbor:
                if not G.has_edge(prev_hop, neighbor):
                    G.add_edge(prev_hop, neighbor, label=relationship)

        # 更新各层级邻居
        if hop_count not in all_hop_neighbors:
            all_hop_neighbors[hop_count] = []
        all_hop_neighbors[hop_count].append(neighbor)

    # 设置图形布局
    pos = nx.spring_layout(G, k=1.2)
    fig, ax = plt.subplots(figsize=(20, 14))

    # 节点颜色
    label_colors = {
        "center": "#FFD700",  # 黄金色
        "neighbor": "#87CEFA",  # 淡蓝色
        "multi_hop_neighbor": "#32CD32",  # 多跳邻居，绿色
        "default": "#D3D3D3"  # 默认灰色
    }

    # 节点颜色映射
    node_colors = [
        label_colors.get(G.nodes[n].get("label", "default"), "#D3D3D3") for n in G.nodes()
    ]
    nx.draw_networkx_nodes(G, pos, node_color=node_colors, node_size=1200, edgecolors="black", ax=ax)

    # 绘制节点标签
    node_labels = {node: node for node in G.nodes()}
    nx.draw_networkx_labels(G, pos, labels=node_labels, font_size=9, ax=ax)

    # 绘制边
    nx.draw_networkx_edges(G, pos, arrowstyle='-|>', arrowsize=50, edge_color='gray', ax=ax)

    # 绘制边标签（我们将这些关系移到侧边栏）
    edge_labels = {(u, v): d.get("label", "") for u, v, d in G.edges(data=True)}

    # 计算侧边栏位置
    sidebar_x = 1.05  # 侧边栏x位置
    sidebar_y_start = 0.95  # 侧边栏y起始位置
    label_margin = 0.02  # 标签的垂直间距
    max_label_height = 0.85  # 侧边栏最大高度

    # 添加关系标签到侧边栏
    for (u, v), label in edge_labels.items():
        if label:  # 如果有标签，绘制在侧边栏
            ax.text(
                sidebar_x, sidebar_y_start, label, fontsize=8, ha='left', va='top',
                transform=ax.transAxes, color="darkred"
            )
            sidebar_y_start -= label_margin  # 每个标签下移

    # 设置标题
    plt.title(title, fontsize=18)
    plt.axis("off")

    # 确保创建输出目录
    output_dir = os.path.dirname(output_file)
    os.makedirs(output_dir, exist_ok=True)  # 确保目标目录存在

    # 保存图像
    plt.savefig(output_file, dpi=300, bbox_inches='tight')
    plt.close()
    print(f"[SAVED] 图已保存为: {output_file}")

"""
文档内提取双星结构，在原有的基础上扩展多跳节点，最大跳数为5
如果返回的重复实体出现次数正好为2，则其出现的两个三元组分别作为双星的两个星型，这个重复实体作为两个星型各自的中心，并且该重复实体用虚拟连接边连接
如果返回的重复实体出现次数大于等于3，则借助三元组中"edge_properties"作为聚类函数进行聚类，将所有该重复实体出现的三元组按照距离远近聚成两类，分别作为两个双星，然后该重复实体作为两个双星的中心，并按照虚拟连接边连接

def generate_virtual_links(star_uids):
    """
    生成虚拟边，连接所有星型结构的中心节点，表示同一实体
    """
    virtual_edges = []
    for i in range(len(star_uids)):
        for j in range(i + 1, len(star_uids)):
            virtual_edges.append({
                "node_1": {"name": star_uids[i], "label": "VirtualCenter"},
                "node_2": {"name": star_uids[j], "label": "VirtualCenter"},
                "relationship": "same_entity"
            })
    return virtual_edges

def extract_bipartite_stars(graph, central_entities):
    """
    提取双星结构：
    - 如果实体出现次数为2，直接作为双星结构
    - 如果实体出现次数>=3，使用KMeans进行聚类
    """
    stars = []

    for entity, count in central_entities.items():
        # 获取该实体的所有三元组
        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        if count == 2:
            # 如果实体出现次数为2，直接构成双星结构
            star1 = {
                'center': entity,
                'triples': triples[:1],  # 第一组三元组
            }
            star2 = {
                'center': entity,
                'triples': triples[1:],  # 第二组三元组
            }
            # 创建虚拟连接边
            virtual_links = generate_virtual_links([entity])
            stars.append((star1, star2, virtual_links))
        elif count >= 3:
            # 如果实体出现次数>=3，使用KMeans进行聚类
            edge_properties = [entry['edge_properties'] for entry in triples]

            # 提取数值特征：假设edge_properties包含 'text_span' 字段
            X = []
            for prop in edge_properties:
                # 假设prop['text_span']包含 'start' 和 'end'
                text_span = prop.get('text_span', {})
                start = text_span.get('start', 0)  # 默认值为0
                end = text_span.get('end', 0)  # 默认值为0

                # 使用 (start + end) / 2 计算质心
                centroid = (start + end) / 2

                # 使用质心作为特征
                X.append([centroid])

            X = np.array(X)

            # 使用KMeans聚类为2类
            kmeans = KMeans(n_clusters=2, random_state=0).fit(X)
            labels = kmeans.labels_

            star1 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 0],  # 第一类三元组
            }
            star2 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 1],  # 第二类三元组
            }
            # 创建虚拟连接边
            virtual_links = generate_virtual_links([entity])
            stars.append((star1, star2, virtual_links))

    return stars

def simplify_edge_labels_with_llm(relationships, llm):
    system_prompt = (
        "You are a helpful assistant. Your task is to simplify relationship phrases "
        "into short, clear, lowercase labels (e.g., 'works_with', 'married_to', 'located_in'). "
        "Avoid full sentences or long phrases. Use underscores to separate words."
    )

    joined = "\n".join(f"- {rel}" for rel in relationships)
    user_prompt = f"""Simplify the following relationship phrases:

{joined}

Return one simplified label per line in the same order."""

    response = llm.generate(user_message=user_prompt, system_message=system_prompt)
    simplified = [line.strip("- ").strip() for line in response.strip().splitlines() if line.strip()]
    return simplified

def remove_single_hop_neighbors(star):
    """ 删除多跳邻居中的跳数为1的邻居，并删除与 neighbors 中节点重复的 multi_hop_neighbors 节点 """

    # 删除 hop_count 为 1 的邻居
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"] if neighbor_info["hop_count"] > 1
    ]

    # 获取 neighbors 中的所有节点
    neighbor_nodes = set(star["neighbors"])

    # 遍历 multi_hop_neighbors 删除重复的节点
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"]
        if neighbor_info["neighbor"] not in neighbor_nodes
    ]

    # 返回修改后的星型结构
    return star



def get_multi_hop_neighbors(node, graph, max_hops, current_hop=1, visited=set(), multi_hop_neighbors=[]):
    """
    递归获取某个节点的多跳邻居，最多为 max_hops 跳。
    """
    if current_hop > max_hops:
        return visited, multi_hop_neighbors

    visited.add(node)

    # 查找该节点的所有三元组，找到与节点连接的其他节点
    for entry in graph:
        if entry['node_1']['name'] == node:
            neighbor = entry['node_2']['name']
            relationship = entry['relationship']
        elif entry['node_2']['name'] == node:
            neighbor = entry['node_1']['name']
            relationship = entry['relationship']
        else:
            continue

        # 如果邻居没有访问过，则递归查找该邻居的多跳邻居
        if neighbor not in visited:
            # 如果没有 'edge' 字段，设置一个默认值
            edge = entry.get('relationship', 'unknown_edge')  # 使用关系作为 edge
            multi_hop_neighbors.append({
                "neighbor": neighbor,
                "hop_count": current_hop,
                "relationship": relationship
            })
            visited, multi_hop_neighbors = get_multi_hop_neighbors(neighbor, graph, max_hops, current_hop + 1, visited, multi_hop_neighbors)

    return visited, multi_hop_neighbors



def modify_center_and_node(center, triples, suffix):
    """
    修改 center 和 node_1.name 中的字段值，给所有字段的值添加后缀（_1 或 _2）

    if isinstance(center, str):

    for triple in triples:

def extend_bipartite_stars_with_hops(graph, central_entities, max_hops=5):
    """
    提取双星结构并延伸多跳邻居
    """
    stars = []

    for entity, count in central_entities.items():
        # 获取该实体的所有三元组
        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        # 1跳邻居（基础的双星结构）
        star1 = {
            'center': entity + "_1",  # 加上后缀 _1
            'triples': triples[:1],  # 第一组三元组
            'neighbors': [entry['node_2']['name'] for entry in triples[:1]],  # 仅选择第一组三元组的邻居
            'edges': [entry['relationship'] for entry in triples[:1]],
            'multi_hop_neighbors': []  # 初始化为空列表
        }

        star2 = {
            'center': entity + "_2",  # 加上后缀 _2
            'triples': triples[1:],  # 第二组三元组
            'neighbors': [entry['node_2']['name'] for entry in triples[1:]],  # 仅选择第二组三元组的邻居
            'edges': [entry['relationship'] for entry in triples[1:]],
            'multi_hop_neighbors': []  # 初始化为空列表
        }

        # 计算多跳邻居
        visited, multi_hop_neighbors = get_multi_hop_neighbors(entity, graph, max_hops)

        # 确保多跳邻居是列表，其中的每个元素是字典
        star1['multi_hop_neighbors'] = multi_hop_neighbors
        star2['multi_hop_neighbors'] = multi_hop_neighbors

        # 删除单跳邻居
        star1 = remove_single_hop_neighbors(star1)
        star2 = remove_single_hop_neighbors(star2)

        # 创建虚拟连接边
        virtual_links = generate_virtual_links([entity + "_1", entity + "_2"])

        # 将双星结构添加到结果列表中
        stars.append((star1, star2, virtual_links))

    return stars




def visualize_star_structure_with_hops(json_data, output_file, title="Star Structure", llm=None):
    """ 可视化包含多跳邻居的星型结构，并限制下一跳节点只能作为出边，同时将关系放置在侧边栏 """
    # 检查 json_data 是否是字典类型，并提取中心节点、邻居节点、边和多跳邻居
    if isinstance(json_data, dict):
        center = json_data.get("center")
        neighbors = json_data.get("neighbors", [])
        edges = json_data.get("edges", [])
        multi_hop_neighbors = json_data.get("multi_hop_neighbors", [])
    else:
        print("Error: json_data should be a dictionary with 'center', 'neighbors', and 'multi_hop_neighbors' keys.")
        return

    if not center or not neighbors or not edges:
        print("Error: Missing required data in json_data. Ensure 'center', 'neighbors', and 'edges' are present.")
        return

    # 使用 LLM 简化边的标签（如果提供了 LLM）
    if llm:
        simplified_edge_labels = simplify_edge_labels_with_llm(edges, llm)
    else:
        simplified_edge_labels = edges  # 如果没有 LLM，则使用原始边标签

    # 创建一个有向图
    G = nx.DiGraph()

    # 添加中心节点和邻居节点
    G.add_node(center, label="center")
    for neighbor in neighbors:
        G.add_node(neighbor, label="neighbor")

    # 添加边，假设每条边的关系为边标签
    for simplified_edge, neighbor in zip(simplified_edge_labels, neighbors):
        G.add_edge(center, neighbor, label=simplified_edge)

    # 记录多跳邻居的前一跳邻居
    prev_neighbors = neighbors  # 初始化为直接邻居节点
    all_hop_neighbors = {0: neighbors}  # 用于存储各层级的邻居

    # 添加多跳邻居并与前一跳邻居相连，确保下一跳节点仅为出边
    for multi_hop in multi_hop_neighbors:
        # 使用字典的键来访问 neighbor、hop_count 和 relationship
        neighbor = multi_hop["neighbor"]
        hop_count = multi_hop["hop_count"]
        relationship = multi_hop["relationship"]

        # 推导出上一跳邻居（基于当前 hop_count 和前一跳的邻居）
        if hop_count > 1 and hop_count - 1 in all_hop_neighbors:
            previous_hop_neighbor = all_hop_neighbors[hop_count - 1]
        else:
            previous_hop_neighbor = neighbors  # 如果是第一跳，则直接连到中心节点

        # 新增多跳邻居节点
        G.add_node(neighbor, label=f"hop_{hop_count}")

        # 确保多跳邻居与前一跳邻居相连，但不允许反向连接
        if hop_count - 1 in all_hop_neighbors:
            prev_neighbors = all_hop_neighbors[hop_count - 1]
            for prev_neighbor in prev_neighbors:
                # 只添加出边，不添加入边
                if not G.has_edge(prev_neighbor, neighbor):  # 确保没有重复的边
                    G.add_edge(prev_neighbor, neighbor, label=relationship)

        # 连接上一跳邻居到当前多跳邻居
        if previous_hop_neighbor != center:  # 确保上一跳不是中心节点
            for prev_hop in previous_hop_neighbor:
                if not G.has_edge(prev_hop, neighbor):
                    G.add_edge(prev_hop, neighbor, label=relationship)

        # 更新各层级邻居
        if hop_count not in all_hop_neighbors:
            all_hop_neighbors[hop_count] = []
        all_hop_neighbors[hop_count].append(neighbor)

    # 设置图形布局
    pos = nx.spring_layout(G, k=1.2)
    fig, ax = plt.subplots(figsize=(20, 14))

    # 节点颜色
    label_colors = {
        "center": "#FFD700",  # 黄金色

def visualize_dual_star_structure(dual_star, output_file, title="Dual-Star Structure", llm=None):
    """
    可视化两个星型结构，并连接它们的虚拟连接边
    """
    # 获取两个星型结构
    star1, star2, virtual_links = dual_star

    # 创建一个有向图
    G = nx.DiGraph()

    # 1. 添加第一个星型的节点和边
    center1 = star1['center']
    for entry in star1['triples']:
        node1 = entry['node_1']['name']
        node2 = entry['node_2']['name']
        relationship = entry['relationship']

        # 添加节点
        G.add_node(node1, label=entry['node_1']['label'])
        G.add_node(node2, label=entry['node_2']['label'])

        # 添加边
        G.add_edge(node1, node2, label=relationship)

    # 2. 添加第二个星型的节点和边
    center2 = star2['center']
    for entry in star2['triples']:
        node1 = entry['node_1']['name']
        node2 = entry['node_2']['name']
        relationship = entry['relationship']

        # 添加节点
        G.add_node(node1, label=entry['node_1']['label'])
        G.add_node(node2, label=entry['node_2']['label'])

        # 添加边
        G.add_edge(node1, node2, label=relationship)

    # 3. 添加虚拟连接边
    for virtual_link in virtual_links:
        node1 = virtual_link['node_1']['name']
        node2 = virtual_link['node_2']['name']
        relationship = virtual_link['relationship']

        # 添加虚拟连接边
        G.add_edge(node1, node2, label=relationship, color='red')

    # 4. 添加多跳邻居
    for star in [star1, star2]:
        for multi_hop in star.get('multi_hop_neighbors', []):
            neighbor = multi_hop["neighbor"]
            hop_count = multi_hop["hop_count"]
            relationship = multi_hop["relationship"]

            if not G.has_node(neighbor):
                G.add_node(neighbor, label=f"hop_{hop_count}")

            for prev_hop in star['neighbors']:
                G.add_edge(prev_hop, neighbor, label=relationship, color='green')

    # 5. 设置图形布局
    pos = nx.spring_layout(G, k=1.2)

    # 6. 创建图形并绘制
    fig, ax = plt.subplots(figsize=(20, 14))

    # 节点颜色
    label_colors = {
        "center": "#FFD700",  # 黄金色

        "default": "#D3D3D3"  # 默认灰色

    ]

如果返回的重复实体出现次数大于等于3，则借助三元组中"edge_properties"作为聚类函数进行聚类，将所有该重复实体出现的三元组按照距离远近聚成两类，分别作为两个双星，然后该重复实体作为两个双星的中心，并按照虚拟连接边连接

def extract_bipartite_stars(graph, central_entities):
    """
    提取双星结构：
    - 如果实体出现次数为2，直接作为双星结构
    - 如果实体出现次数>=3，使用KMeans进行聚类
    """
    stars = []

    for entity, count in central_entities.items():
        # 获取该实体的所有三元组
        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        if count == 2:
            # 如果实体出现次数为2，直接构成双星结构
            star1 = {
                'center': entity,
                'triples': triples[:1],  # 第一组三元组
            }
            star2 = {
                'center': entity,
                'triples': triples[1:],  # 第二组三元组
            }
            # 创建虚拟连接边
            stars.append((star1, star2))
        elif count >= 3:
            # 如果实体出现次数>=3，使用KMeans进行聚类
            edge_properties = [entry['edge_properties'] for entry in triples]

            # 提取数值特征：假设edge_properties包含 'text_span' 字段
            X = []
            for prop in edge_properties:
                # 假设prop['text_span']包含 'start' 和 'end'
                text_span = prop.get('text_span', {})
                start = text_span.get('start', 0)  # 默认值为0
                end = text_span.get('end', 0)  # 默认值为0

                # 使用 (start + end) / 2 计算质心
                centroid = (start + end) / 2

                # 使用质心作为特征
                X.append([centroid])

            X = np.array(X)

            # 使用KMeans聚类为2类
            kmeans = KMeans(n_clusters=2, random_state=0).fit(X)
            labels = kmeans.labels_

            star1 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 0],  # 第一类三元组
            }
            star2 = {
                'center': entity,
                'triples': [triples[i] for i in range(len(triples)) if labels[i] == 1],  # 第二类三元组
            }
            # 创建虚拟连接边
            stars.append((star1, star2))

    return stars

def simplify_edge_labels_with_llm(relationships, llm):
    system_prompt = (
        "You are a helpful assistant. Your task is to simplify relationship phrases "
        "into short, clear, lowercase labels (e.g., 'works_with', 'married_to', 'located_in'). "
        "Avoid full sentences or long phrases. Use underscores to separate words."
    )

    joined = "\n".join(f"- {rel}" for rel in relationships)
    user_prompt = f"""Simplify the following relationship phrases:

{joined}

Return one simplified label per line in the same order."""

    response = llm.generate(user_message=user_prompt, system_message=system_prompt)
    simplified = [line.strip("- ").strip() for line in response.strip().splitlines() if line.strip()]
    return simplified

def remove_single_hop_neighbors(star):
    """ 删除多跳邻居中的跳数为1的邻居，并删除与 neighbors 中节点重复的 multi_hop_neighbors 节点 """

    # 删除 hop_count 为 1 的邻居
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"] if neighbor_info["hop_count"] > 1
    ]

    # 获取 neighbors 中的所有节点
    neighbor_nodes = set(star["neighbors"])

    # 遍历 multi_hop_neighbors 删除重复的节点
    star["multi_hop_neighbors"] = [
        neighbor_info for neighbor_info in star["multi_hop_neighbors"]
        if neighbor_info["neighbor"] not in neighbor_nodes
    ]

    # 返回修改后的星型结构
    return star



def get_multi_hop_neighbors(node, graph, max_hops, current_hop=1, visited=set(), multi_hop_neighbors=[]):
    """
    递归获取某个节点的多跳邻居，最多为 max_hops 跳。
    """
    if current_hop > max_hops:
        return visited, multi_hop_neighbors

    visited.add(node)

    # 查找该节点的所有三元组，找到与节点连接的其他节点
    for entry in graph:
        if entry['node_1']['name'] == node:
            neighbor = entry['node_2']['name']
            relationship = entry['relationship']
        elif entry['node_2']['name'] == node:
            neighbor = entry['node_1']['name']
            relationship = entry['relationship']
        else:
            continue

        # 如果邻居没有访问过，则递归查找该邻居的多跳邻居
        if neighbor not in visited:
            # 如果没有 'edge' 字段，设置一个默认值
            edge = entry.get('relationship', 'unknown_edge')  # 使用关系作为 edge
            multi_hop_neighbors.append({
                "neighbor": neighbor,
                "hop_count": current_hop,
                "relationship": relationship
            })
            visited, multi_hop_neighbors = get_multi_hop_neighbors(neighbor, graph, max_hops, current_hop + 1, visited, multi_hop_neighbors)

    return visited, multi_hop_neighbors



def modify_center_and_node(center, triples, suffix):

def extend_bipartite_stars_with_hops(graph, central_entities, max_hops=5):
    """
    提取双星结构并延伸多跳邻居
    """
    stars = []

    for entity, count in central_entities.items():

        triples = [entry for entry in graph if entry['node_1']['name'] == entity]

        # 1跳邻居（基础的双星结构）
        star1 = {
            'center': entity + "_1",  # 加上后缀 _1
            'triples': triples[:1],  # 第一组三元组
            'neighbors': [entry['node_2']['name'] for entry in triples[:1]],  # 仅选择第一组三元组的邻居
            'edges': [entry['relationship'] for entry in triples[:1]],

            'neighbors': [entry['node_2']['name'] for entry in triples[1:]],  # 仅选择第二组三元组的邻居

        # 删除单跳邻居
        star1 = remove_single_hop_neighbors(star1)
        star2 = remove_single_hop_neighbors(star2)

        # 将双星结构添加到结果列表中
        stars.append((star1, star2))

    return stars




def visualize_star_structure_with_hops(json_data, output_file, title="Star Structure", llm=None):
    """ 可视化包含多跳邻居的星型结构，并限制下一跳节点只能作为出边，同时将关系放置在侧边栏 """
    # 检查 json_data 是否是字典类型，并提取中心节点、邻居节点、边和多跳邻居
    if isinstance(json_data, dict):
        center = json_data.get("center")
        neighbors = json_data.get("neighbors", [])
        edges = json_data.get("edges", [])
        multi_hop_neighbors = json_data.get("multi_hop_neighbors", [])
    else:
        print("Error: json_data should be a dictionary with 'center', 'neighbors', and 'multi_hop_neighbors' keys.")
        return

    if not center or not neighbors or not edges:
        print("Error: Missing required data in json_data. Ensure 'center', 'neighbors', and 'edges' are present.")
        return

    # 使用 LLM 简化边的标签（如果提供了 LLM）
    if llm:
        simplified_edge_labels = simplify_edge_labels_with_llm(edges, llm)
    else:
        simplified_edge_labels = edges  # 如果没有 LLM，则使用原始边标签

    # 创建一个有向图
    G = nx.DiGraph()

    # 添加中心节点和邻居节点
    G.add_node(center, label="center")
    for neighbor in neighbors:
        G.add_node(neighbor, label="neighbor")

    # 添加边，假设每条边的关系为边标签
    for simplified_edge, neighbor in zip(simplified_edge_labels, neighbors):
        G.add_edge(center, neighbor, label=simplified_edge)

    # 记录多跳邻居的前一跳邻居
    prev_neighbors = neighbors  # 初始化为直接邻居节点
    all_hop_neighbors = {0: neighbors}  # 用于存储各层级的邻居

    # 添加多跳邻居并与前一跳邻居相连，确保下一跳节点仅为出边
    for multi_hop in multi_hop_neighbors:
        # 使用字典的键来访问 neighbor、hop_count 和 relationship
        neighbor = multi_hop["neighbor"]
        hop_count = multi_hop["hop_count"]
        relationship = multi_hop["relationship"]

        # 推导出上一跳邻居（基于当前 hop_count 和前一跳的邻居）
        if hop_count > 1 and hop_count - 1 in all_hop_neighbors:
            previous_hop_neighbor = all_hop_neighbors[hop_count - 1]
        else:
            previous_hop_neighbor = neighbors  # 如果是第一跳，则直接连到中心节点

        # 新增多跳邻居节点
        G.add_node(neighbor, label=f"hop_{hop_count}")

        # 确保多跳邻居与前一跳邻居相连，但不允许反向连接
        if hop_count - 1 in all_hop_neighbors:
            prev_neighbors = all_hop_neighbors[hop_count - 1]
            for prev_neighbor in prev_neighbors:
                # 只添加出边，不添加入边
                if not G.has_edge(prev_neighbor, neighbor):  # 确保没有重复的边
                    G.add_edge(prev_neighbor, neighbor, label=relationship)

        # 连接上一跳邻居到当前多跳邻居
        if previous_hop_neighbor != center:  # 确保上一跳不是中心节点
            for prev_hop in previous_hop_neighbor:
                if not G.has_edge(prev_hop, neighbor):
                    G.add_edge(prev_hop, neighbor, label=relationship)

        # 更新各层级邻居
        if hop_count not in all_hop_neighbors:
            all_hop_neighbors[hop_count] = []
        all_hop_neighbors[hop_count].append(neighbor)

    # 设置图形布局
    pos = nx.spring_layout(G, k=1.2)
    fig, ax = plt.subplots(figsize=(20, 14))

    # 节点颜色
    label_colors = {
        "center": "#FFD700",  # 黄金色
        "neighbor": "#87CEFA",  # 淡蓝色
        "multi_hop_neighbor": "#32CD32",  # 多跳邻居，绿色
        "default": "#D3D3D3"  # 默认灰色
    }

    # 节点颜色映射
    node_colors = [
        label_colors.get(G.nodes[n].get("label", "default"), "#D3D3D3") for n in G.nodes()
    ]
    nx.draw_networkx_nodes(G, pos, node_color=node_colors, node_size=1200, edgecolors="black", ax=ax)

    # 绘制节点标签
    node_labels = {node: node for node in G.nodes()}
    nx.draw_networkx_labels(G, pos, labels=node_labels, font_size=9, ax=ax)

    # 绘制边
    nx.draw_networkx_edges(G, pos, arrowstyle='-|>', arrowsize=50, edge_color='gray', ax=ax)

    # 绘制边标签（我们将这些关系移到侧边栏）
    edge_labels = {(u, v): d.get("label", "") for u, v, d in G.edges(data=True)}

    # 计算侧边栏位置
    sidebar_x = 1.05  # 侧边栏x位置
    sidebar_y_start = 0.95  # 侧边栏y起始位置
    label_margin = 0.02  # 标签的垂直间距
    max_label_height = 0.85  # 侧边栏最大高度

    # 添加关系标签到侧边栏
    for (u, v), label in edge_labels.items():
        if label:  # 如果有标签，绘制在侧边栏
            ax.text(
                sidebar_x, sidebar_y_start, label, fontsize=8, ha='left', va='top',
                transform=ax.transAxes, color="darkred"
            )
            sidebar_y_start -= label_margin  # 每个标签下移

    # 设置标题
    plt.title(title, fontsize=18)
    plt.axis("off")

    # 确保创建输出目录
    output_dir = os.path.dirname(output_file)
    os.makedirs(output_dir, exist_ok=True)  # 确保目标目录存在

    # 保存图像
    plt.savefig(output_file, dpi=300, bbox_inches='tight')
    plt.close()
    print(f"[SAVED] 图已保存为: {output_file}")


def visualize_dual_star_structure(dual_star, output_file, title="Dual-Star Structure", llm=None):
    """
    可视化两个星型结构，并连接它们的虚拟连接边
    """
    # 获取两个星型结构
    star1, star2 = dual_star

    # 创建一个有向图
    G = nx.DiGraph()

    # 1. 添加第一个星型的节点和边
    center1 = star1['center']
    for entry in star1['triples']:
        node1 = entry['node_1']['name']
        node2 = entry['node_2']['name']
        relationship = entry['relationship']

        # 添加节点
        G.add_node(node1, label=entry['node_1']['label'])
        G.add_node(node2, label=entry['node_2']['label'])

        # 添加边
        G.add_edge(node1, node2, label=relationship)

    # 2. 添加第二个星型的节点和边
    center2 = star2['center']
    for entry in star2['triples']:
        node1 = entry['node_1']['name']
        node2 = entry['node_2']['name']
        relationship = entry['relationship']

        # 添加节点
        G.add_node(node1, label=entry['node_1']['label'])
        G.add_node(node2, label=entry['node_2']['label'])

        # 添加边
        G.add_edge(node1, node2, label=relationship)

    # 3. 设置图形布局
    pos = nx.spring_layout(G, k=1.2)

    # 4. 创建图形并绘制
    fig, ax = plt.subplots(figsize=(20, 14))

    # 节点颜色
    label_colors = {
        "center": "#FFD700",  # 黄金色
        "neighbor": "#87CEFA",  # 淡蓝色
        "multi_hop_neighbor": "#32CD32",  # 多跳邻居，绿色
        "default": "#D3D3D3"  # 默认灰色
    }

    # 节点颜色映射
    node_colors = [
        label_colors.get(G.nodes[n].get("label", "default"), "#D3D3D3") for n in G.nodes()
    ]
    nx.draw_networkx_nodes(G, pos, node_color=node_colors, node_size=1200, edgecolors="black", ax=ax)

    # 绘制节点标签
    node_labels = {node: node for node in G.nodes()}
    nx.draw_networkx_labels(G, pos, labels=node_labels, font_size=9, ax=ax)

    # 绘制边
    edge_colors = [d.get('color', 'gray') for u, v, d in G.edges(data=True)]
    nx.draw_networkx_edges(G, pos, edge_color=edge_colors, width=2, ax=ax)

    # 绘制边标签（关系标签）
    edge_labels = {(u, v): d.get("label", "") for u, v, d in G.edges(data=True)}
    nx.draw_networkx_edge_labels(G, pos, edge_labels=edge_labels, font_size=7, font_color='darkred', ax=ax)

    # 设置标题
    plt.title(title, fontsize=18)
    plt.axis("off")

    # 确保创建输出目录
    output_dir = os.path.dirname(output_file)
    os.makedirs(output_dir, exist_ok=True)  # 确保目标目录存在

    # 保存图像
    plt.savefig(output_file, dpi=300, bbox_inches='tight')
    plt.close()
    print(f"[SAVED] 图已保存为: {output_file}")
